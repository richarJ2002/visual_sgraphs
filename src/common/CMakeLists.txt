# The private sources have their own list so the unit tests can compile them
# without the ROS-facing common.cc.
set(VS_GRAPHS_FRONTEND_COMMON_PRIVATE_SOURCES
  private_functions/keepFirstPointPerVoxel.cc
)
list(TRANSFORM VS_GRAPHS_FRONTEND_COMMON_PRIVATE_SOURCES
  PREPEND "${CMAKE_CURRENT_LIST_DIR}/")

set(VS_GRAPHS_FRONTEND_COMMON_SOURCES
  public_functions/common.cc
)
list(TRANSFORM VS_GRAPHS_FRONTEND_COMMON_SOURCES
  PREPEND "${CMAKE_CURRENT_LIST_DIR}/")
list(APPEND VS_GRAPHS_FRONTEND_COMMON_SOURCES
  ${VS_GRAPHS_FRONTEND_COMMON_PRIVATE_SOURCES})
